💻 快速上手示例

Python - 对话 API 调用示例
import requests # TRAE API 配置 API_URL = "https://api.trae.cn/v2/chat" API_KEY = "your-api-key-here" def ask_trae(prompt: str) -> dict: """向 TRAE AI 发送编程问题""" headers = { "Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json" } payload = { "model": "trae-pro", "messages": [ {"role": "user", "content": prompt} ] } response = requests.post(API_URL, json=payload, headers=headers) return response.json() # 使用示例 result = ask_trae("用 Python 写一个快速排序函数") print(result["choices"][0]["message"]["content"])
